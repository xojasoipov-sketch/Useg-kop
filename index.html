⚡ One Endpoint, All Providers

Drop-in replacement for OpenAI SDK — just change the base URL.

// Install: npm install omniroute

import OmniRoute from 'omniroute';

const client = new OmniRoute({
  baseURL: 'https://omniroute.online/v1',
  routing: 'cost-optimized', // or 'priority', 'weighted', 'fusion'
  compression: 'rtk+caveman', // 15–95% token reduction
  freeFirst: true, // use free providers first
});

const response = await client.chat.completions.create({
  model: 'auto', // OmniRoute picks the best available
  messages: [{ role: 'user', content: 'Hello!' }],
});
